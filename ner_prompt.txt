Extract entities and relationships from police notes.

You have the redacted police note and a list of people involved with the stop; identify which candidate person is most likely each person in the stop. Even if you cannot identify which candidate is which, you can still report the relationship between them

Read and interpret free-text police notes. infer unknown information, you're a superintelligence

Identify and extract these entities:
people
location: residences, destinations, businesses, institutions, and secondary incident locations.
vehicle: needed mainly to connect drivers, passengers, owners, and renters.
item: weapons, drugs, cash, paraphernalia, stolen property, documents, bicycles.
organization: gangs, street groups, employers, hospitals, courts, schools.

import { z } from "zod";

const supportSchema = z.object({
  evidence: z.string(),
  source: z.enum([
    "officer_observation",
    "person_statement",
    "third_party_statement",
    "database_record",
    "unknown",
  ]),
  inferred: z.boolean().default(false),
  negated: z.boolean().default(false),
});

const entitySchema = z.object({
  id: z.string(),
  type: z.enum([
    "person",
    "location",
    "vehicle",
    "item",
    "organization",
  ]),
  label: z.string(),
  subtype: z.string().nullable(),
  candidatePersonId: z.string().nullable().optional(),
  candidateConfidence: z.number().min(0).max(1).optional(),
  roles: z.array(z.enum([
    "contact_subject",
    "driver",
    "passenger",
    "victim",
    "suspect",
    "caller",
    "witness",
    "owner",
    "other",
  ])).default([]),
});

export const caseNoteSchema = z.object({
  fcNum: z.string(),

  entities: z.array(entitySchema),

  personFacts: z.array(z.object({
    personId: z.string(),
    category: z.enum([
      "identity",
      "appearance",
      "legal",
      "affiliation",
      "health",
      "housing",
      "behavior",
      "other",
    ]),
    value: z.string(),
    support: supportSchema,
  })),

  relationships: z.array(z.object({
    fromId: z.string(),
    toId: z.string(),
    kind: z.enum([
      "personal",
      "affiliation",
      "located_at",
      "occupant_of",
      "owns_or_controls",
      "possesses",
      "interaction",
      "other",
    ]),
    label: z.string(),
    support: supportSchema,
  })),

  outcomes: z.array(z.object({
    personId: z.string(),
    action: z.enum([
      "identified",
      "queried",
      "frisked",
      "searched",
      "warned",
      "cited",
      "trespassed",
      "released",
      "arrested",
      "summonsed",
      "transported",
      "other",
    ]),
    detail: z.string().nullable(),
    support: supportSchema,
  })),
});
